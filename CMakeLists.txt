# AppDimens for LVGL — works as a plain CMake library, an ESP-IDF component and a Zephyr module.
if(ESP_PLATFORM)
  idf_component_register(SRCS "src/lv_appdimens.c" INCLUDE_DIRS "src" REQUIRES lvgl__lvgl)
  target_compile_definitions(${COMPONENT_LIB} PUBLIC LV_LVGL_H_INCLUDE_SIMPLE)
  if(CONFIG_APPDIMENS_NO_LIBM)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC AD_NO_LIBM)
  endif()
  return()
endif()

if(ZEPHYR_BASE)
  zephyr_library()
  zephyr_library_sources(src/lv_appdimens.c)
  zephyr_include_directories(src)
  zephyr_compile_definitions(LV_LVGL_H_INCLUDE_SIMPLE)
  return()
endif()

cmake_minimum_required(VERSION 3.13)
project(appdimens_lvgl VERSION 3.2.0 LANGUAGES C)
add_library(appdimens_lvgl STATIC src/lv_appdimens.c)
target_include_directories(appdimens_lvgl PUBLIC src)
target_compile_definitions(appdimens_lvgl PUBLIC LV_LVGL_H_INCLUDE_SIMPLE)
if(TARGET lvgl)
  target_link_libraries(appdimens_lvgl PUBLIC lvgl)
elseif(TARGET lvgl::lvgl)
  target_link_libraries(appdimens_lvgl PUBLIC lvgl::lvgl)
endif()
option(APPDIMENS_NO_LIBM "Use the built-in log/pow (no libm)" OFF)
if(APPDIMENS_NO_LIBM)
  target_compile_definitions(appdimens_lvgl PUBLIC AD_NO_LIBM)
else()
  find_library(M_LIB m)
  if(M_LIB)
    target_link_libraries(appdimens_lvgl PUBLIC ${M_LIB})
  endif()
endif()
add_library(appdimens::lvgl ALIAS appdimens_lvgl)
